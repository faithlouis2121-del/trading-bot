async function executeCommand() { const inputField = document.getElementById('command-input'); const logBox = document.getElementById('terminal-logs'); const command = inputField.value.trim(); if (!command) return; // Log user command locally immediately logBox.innerHTML += `[${new Date().toLocaleTimeString()}] > ${command}
`; inputField.value = ''; try { const response = await fetch('/api/command', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ command }) }); const data = await response.json(); if (data.status === 'success') { logBox.innerHTML += `[${new Date().toLocaleTimeString()}] ${data.reply}
`; } else { logBox.innerHTML += `[${new Date().toLocaleTimeString()}] Error processing command.
`; } } catch (err) { logBox.innerHTML += `[${new Date().toLocaleTimeString()}] Network error reaching command server.
`; } logBox.scrollTop = logBox.scrollHeight; }
